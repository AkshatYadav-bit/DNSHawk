import queue
import threading
import tkinter as tk

from dnshawk.collector.tshark_capture import TSharkCapture
from dnshawk.parser.dns_parser import parse_tshark_line
from dnshawk.models.dns_window import DNSWindow
from dnshawk.features.feature_engine import FeatureEngine
from dnshawk.aggregation.window import WindowManager
from dnshawk.detection.rule_detector import RuleDetector

from dnshawk.gui.feature_panel import FeaturePanel
from dnshawk.gui.kali_test import KaliTestPanel
from dnshawk.gui.header import HeaderPanel
from dnshawk.gui.controls import ControlPanel
from dnshawk.gui.event_panel import EventPanel
from dnshawk.gui.window_panel import WindowPanel
from dnshawk.gui.feature_panel import FeaturePanel


class DNSDashboard:
    """
    Main DNSHawk dashboard coordinator.

    Responsibilities:
    - Capture lifecycle
    - Event queue
    - DNS window state
    - Feature extraction
    - Rule-based detection
    - Communication between GUI panels
    """

    def __init__(self, root):

        self.root = root

        self.root.title(
            "DNSHawk - DNS Traffic Monitor"
        )

        self.root.geometry(
            "1450x750"
        )

        # -------------------------------------------------
        # Capture state
        # -------------------------------------------------

        self.capture = None
        self.capture_thread = None
        self.running = False

        # Queue safely transfers events from
        # capture thread to Tkinter thread.
        self.event_queue = queue.Queue()

        # -------------------------------------------------
        # DNS processing
        # -------------------------------------------------

        self.feature_engine = FeatureEngine()

        self.rule_detector = RuleDetector()

        self.window_manager = WindowManager(
            window_size_seconds=60
        )

        # -------------------------------------------------
        # DNS windows
        # -------------------------------------------------

        # Key:
        # (src_ip, domain, window_start)

        self.windows = {}

        # Key -> detection result
        self.window_detections = {}

        # Currently selected window
        self.selected_window_key = None

        # -------------------------------------------------
        # Statistics
        # -------------------------------------------------

        self.total_events = 0

        # -------------------------------------------------
        # Build GUI
        # -------------------------------------------------

        self.build_interface()

        # Start queue polling
        self.root.after(
            200,
            self.process_events
        )

    # =====================================================
    # BUILD GUI
    # =====================================================

    def build_interface(self):

        # -------------------------------------------------
        # Header
        # -------------------------------------------------

        self.header = HeaderPanel(
            self.root
        )

        # -------------------------------------------------
        # Controls
        # -------------------------------------------------

        self.controls = ControlPanel(
            self.root,
            start_callback=self.start_capture,
            stop_callback=self.stop_capture
        )

        # kali test controls

        self.kali_test = KaliTestPanel(
            self.root,
            start_callback=self.start_kali_test,
            stop_callback=self.stop_kali_test
        )

        self.kali_test.frame.pack(
            fill="x",
            padx=10,
            pady=(0, 5)
        )

        # -------------------------------------------------
        # Main three-panel area
        # -------------------------------------------------

        main = tk.PanedWindow(
            self.root,
            orient="horizontal"
        )

        main.pack(
            fill="both",
            expand=True,
            padx=10,
            pady=10
        )

        # -------------------------------------------------
        # Panel 1: Live DNS Events
        # -------------------------------------------------

        self.event_panel = EventPanel(
            main
        )

        main.add(
            self.event_panel.frame,
            stretch="always"
        )

        # -------------------------------------------------
        # Panel 2: DNS Windows
        # -------------------------------------------------

        self.window_panel = WindowPanel(
            main,
            selection_callback=self.on_window_selected
        )

        main.add(
            self.window_panel.frame,
            stretch="always"
        )

        # -------------------------------------------------
        # Panel 3: Features + Detection
        # -------------------------------------------------

        self.feature_panel = FeaturePanel(
            main
        )

        main.add(
            self.feature_panel.frame,
            stretch="always"
        )

        # -------------------------------------------------
        # Bottom status
        # -------------------------------------------------

        self.event_count_label = tk.Label(
            self.root,
            text="Events captured: 0",
            padx=8,
            pady=5
        )

        self.event_count_label.pack(
            anchor="w"
        )

    # =====================================================
    # START CAPTURE
    # =====================================================

    def start_capture(self):

        if self.running:
            return

        try:

            interface = (
                self.controls
                .get_interface()
            )

        except ValueError:

            self.header.set_status(
                "● INVALID INTERFACE",
                "red"
            )

            return

        self.capture = TSharkCapture(
            interface
        )

        try:

            stream = self.capture.start()

        except Exception as error:

            self.header.set_status(
                f"● ERROR: {error}",
                "red"
            )

            self.capture = None

            return

        self.running = True

        self.controls.set_running()

        self.header.set_status(
            "● CAPTURING",
            "green"
        )

        self.capture_thread = threading.Thread(
            target=self.capture_loop,
            args=(stream,),
            daemon=True
        )

        self.capture_thread.start()

    def start_kali_test(self, interface):

        if self.running:
            return

        self.capture = TSharkCapture(
            interface
        )

        try:
            stream = self.capture.start()

        except Exception as error:

            self.header.set_status(
                f"● KALI TEST ERROR: {error}",
                "red"
            )

            self.capture = None
            return

        self.running = True

        self.kali_test.set_running()

        self.header.set_status(
            "● KALI TEST CAPTURING",
            "green"
        )

        self.capture_thread = threading.Thread(
            target=self.capture_loop,
            args=(stream,),
            daemon=True
        )

        self.capture_thread.start()

    # =====================================================
    # STOP CAPTURE
    # =====================================================

    def stop_capture(self):

        if not self.running:
            return

        self.running = False

        if self.capture is not None:
            self.capture.stop()

        self.capture = None

        self.controls.set_stopped()
        self.kali_test.set_stopped()

        self.header.set_status(
            "● STOPPED",
            "red"
        )

    def stop_kali_test(self):

        if not self.running:
            return

        self.running = False

        if self.capture is not None:
            self.capture.stop()

        self.capture = None

        self.kali_test.set_stopped()

        self.header.set_status(
            "● KALI TEST STOPPED",
            "red"
        )

    # =====================================================
    # BACKGROUND CAPTURE THREAD
    # =====================================================

    def capture_loop(self, stream):

        try:

            for line in stream:

                if not self.running:
                    break

                event = parse_tshark_line(
                    line
                )

                if event is not None:

                    self.event_queue.put(
                        event
                    )

        except Exception:
            # Capture thread should not crash
            # the GUI.
            pass

    # =====================================================
    # PROCESS EVENT QUEUE
    # =====================================================

    def process_events(self):

        while True:

            try:

                event = (
                    self.event_queue
                    .get_nowait()
                )

            except queue.Empty:

                break

            self.handle_event(
                event
            )

        self.root.after(
            200,
            self.process_events
        )

    # =====================================================
    # HANDLE ONE DNS EVENT
    # =====================================================

    def handle_event(self, event):

        self.total_events += 1

        # -------------------------------------------------
        # 1. Add event to live event panel
        # -------------------------------------------------

        self.event_panel.insert_event(
            event
        )

        # -------------------------------------------------
        # 2. Determine fixed 60-second window
        # -------------------------------------------------

        window_start = (
            self.window_manager
            .get_window_start(
                event.timestamp
            )
        )

        # -------------------------------------------------
        # 3. Build unique window key
        # -------------------------------------------------

        key = (
            event.src_ip,
            event.domain,
            window_start
        )

        # -------------------------------------------------
        # 4. Create window if needed
        # -------------------------------------------------

        if key not in self.windows:

            self.windows[key] = DNSWindow(
                src_ip=event.src_ip,
                domain=event.domain,
                start_time=window_start,
                window_size_seconds=60
            )

            self.window_panel.add_window(
                key,
                self.windows[key]
            )

            # Detection does not exist yet.
            self.window_detections[key] = None

        # -------------------------------------------------
        # 5. Add event to correct window
        # -------------------------------------------------

        window = self.windows[key]

        window.add_event(
            event
        )

        # -------------------------------------------------
        # 6. Extract features
        # -------------------------------------------------

        features = (
            self.feature_engine
            .extract(window)
        )

        # -------------------------------------------------
        # 7. Run rule-based detection
        # -------------------------------------------------

        detection = (
            self.rule_detector
            .assess(features)
        )

        self.window_detections[key] = (
            detection
        )

        # -------------------------------------------------
        # 8. Update window panel
        # -------------------------------------------------

        self.window_panel.update_window(
            key,
            window,
            detection
        )

        # -------------------------------------------------
        # 9. Automatically select first window
        # -------------------------------------------------

        if self.selected_window_key is None:

            self.selected_window_key = key

            self.window_panel.select_window(
                key
            )

        # -------------------------------------------------
        # 10. Refresh selected window
        # -------------------------------------------------

        if self.selected_window_key == key:

            self.update_selected_window()

        # -------------------------------------------------
        # 11. Update event count
        # -------------------------------------------------

        self.event_count_label.config(
            text=(
                f"Events captured: "
                f"{self.total_events}"
            )
        )

    # =====================================================
    # WINDOW SELECTION
    # =====================================================

    def on_window_selected(self, key):

        if key not in self.windows:
            return

        self.selected_window_key = key

        self.update_selected_window()

    # =====================================================
    # UPDATE SELECTED WINDOW
    # =====================================================

    def update_selected_window(self):

        if self.selected_window_key is None:
            return

        if self.selected_window_key not in self.windows:
            return

        window = self.windows[
            self.selected_window_key
        ]

        # -------------------------------------------------
        # Extract current features
        # -------------------------------------------------

        features = (
            self.feature_engine
            .extract(window)
        )

        # -------------------------------------------------
        # Get current detection
        # -------------------------------------------------

        detection = (
            self.window_detections
            .get(
                self.selected_window_key
            )
        )

        # -------------------------------------------------
        # Update feature panel
        # -------------------------------------------------

        self.feature_panel.update(
            window,
            features,
            detection
        )

    # =====================================================
    # CLOSE APPLICATION
    # =====================================================

    def close_application(self):

        self.running = False

        if self.capture is not None:

            self.capture.stop()

            self.capture = None

        self.root.destroy()


# =========================================================
# MAIN
# =========================================================

def main():

    root = tk.Tk()

    dashboard = DNSDashboard(
        root
    )

    root.protocol(
        "WM_DELETE_WINDOW",
        dashboard.close_application
    )

    root.mainloop()


if __name__ == "__main__":
    main()

from tkinter import ttk


class ControlPanel:
    """
    Capture control panel.

    Handles:
    - Network interface input
    - Start button
    - Stop button
    """

    def __init__(
        self,
        parent,
        start_callback,
        stop_callback
    ):

        self.frame = ttk.Frame(
            parent,
            padding=(10, 0, 10, 10)
        )

        self.frame.pack(
            fill="x"
        )

        ttk.Label(
            self.frame,
            text="Interface:"
        ).pack(
            side="left"
        )

        self.interface_entry = ttk.Entry(
            self.frame,
            width=8
        )

        self.interface_entry.insert(
            0,
            "5"
        )

        self.interface_entry.pack(
            side="left",
            padx=(5, 10)
        )

        self.start_button = ttk.Button(
            self.frame,
            text="Start Capture",
            command=start_callback
        )

        self.start_button.pack(
            side="left",
            padx=5
        )

        self.stop_button = ttk.Button(
            self.frame,
            text="Stop Capture",
            command=stop_callback,
            state="disabled"
        )

        self.stop_button.pack(
            side="left",
            padx=5
        )

    def get_interface(self):
        """
        Return selected TShark interface as integer.
        """

        return int(
            self.interface_entry
            .get()
            .strip()
        )

    def set_running(self):
        """
        Update controls when capture starts.
        """

        self.start_button.config(
            state="disabled"
        )

        self.stop_button.config(
            state="normal"
        )

        self.interface_entry.config(
            state="disabled"
        )

    def set_stopped(self):
        """
        Update controls when capture stops.
        """

        self.start_button.config(
            state="normal"
        )

        self.stop_button.config(
            state="disabled"
        )

        self.interface_entry.config(
            state="normal"
        )

from tkinter import ttk


class EventPanel:
    """
    Displays live DNS events.
    """

    def __init__(self, parent):

        self.frame = ttk.LabelFrame(
            parent,
            text="Live DNS Events",
            padding=8
        )

        event_columns = (
            "time",
            "src",
            "dst",
            "query",
            "qtype",
            "length"
        )

        self.table = ttk.Treeview(
            self.frame,
            columns=event_columns,
            show="headings"
        )

        event_headings = {
            "time": "Time",
            "src": "Source",
            "dst": "Destination",
            "query": "Query",
            "qtype": "Type",
            "length": "Length"
        }

        event_widths = {
            "time": 80,
            "src": 125,
            "dst": 125,
            "query": 260,
            "qtype": 55,
            "length": 65
        }

        for column in event_columns:

            self.table.heading(
                column,
                text=event_headings[column]
            )

            self.table.column(
                column,
                width=event_widths[column],
                anchor="center"
            )

        scrollbar = ttk.Scrollbar(
            self.frame,
            orient="vertical",
            command=self.table.yview
        )

        self.table.configure(
            yscrollcommand=scrollbar.set
        )

        self.table.pack(
            side="left",
            fill="both",
            expand=True
        )

        scrollbar.pack(
            side="right",
            fill="y"
        )

    def insert_event(self, event):
        """
        Add one DNS event to the table.
        """

        timestamp = event.timestamp.strftime(
            "%H:%M:%S"
        )

        self.table.insert(
            "",
            "end",
            values=(
                timestamp,
                event.src_ip,
                event.dst_ip,
                event.query,
                event.qtype,
                event.packet_length
            )
        )

        # Keep only latest 200 displayed events.
        rows = self.table.get_children()

        if len(rows) > 200:

            self.table.delete(
                rows[0]
            )

        self.table.yview_moveto(1)

             "Queries / Second"),

            ttk.Label(
                feature_container,
                text=label + ":",
                font=("Segoe UI", 9)
            ).grid(
                row=row,
                column=0,
                sticky="w",
                pady=4
            )

            value_label = ttk.Label(
                feature_container,
                text="0",
                font=("Consolas", 9, "bold")
            )

            value_label.grid(
                row=row,
                column=1,
                sticky="e",
                padx=(15, 0),
                pady=4
            )

            self.feature_labels[
                key
            ] = value_label

    def update(
        self,
        window,
        features,
        detection
    ):
        """
        Update selected window information,
        detection status and features.
        """

        # -------------------------------------------------
        # Window information
        # -------------------------------------------------

        self.selected_info_label.config(
            text=(
                f"Source: {window.src_ip}\n"
                f"Domain: {window.domain}\n"
                f"Window: "
                f"{window.start_time.strftime('%H:%M:%S')}"
                f" - "
                f"{window.end_time.strftime('%H:%M:%S')}"
            )
        )

        # -------------------------------------------------
        # Detection information
        # -------------------------------------------------

        if detection is None:

            self.detection_summary_label.config(
                text="Detection: Not evaluated"
            )

            for rule_name, label in (
                self.rule_status_labels.items()
            ):

                display_name = (
                    rule_name
                    .replace("_", " ")
                    .title()
                )

                label.config(
                    text=f"○ {display_name}"
                )

        else:

            triggered_count = len(
                detection["triggered_rules"]
            )

            evidence_text = (
                "MET"
                if detection["minimum_evidence_met"]
                else "NOT MET"
            )

            self.detection_summary_label.config(
                text=(
                    f"Evidence: {evidence_text} | "
                    f"Score: "
                    f"{detection['score']:.1f}/"
                    f"{detection['max_score']:.1f} | "
                    f"Rules: "
                    f"{triggered_count}/5"
                )
            )

            for rule_name, result in (
                detection["rule_results"].items()
            ):

                label = self.rule_status_labels[
                    rule_name
                ]

                display_name = (
                    rule_name
                    .replace("_", " ")
                    .title()
                )

                value = result["value"]
                threshold = result["threshold"]

                if result["triggered"]:

                    label.config(
                        text=(
                            f"✓ {display_name}  "
                            f"({value:.3f} >= "
                            f"{threshold:.3f})"
                        )
                    )

                else:

                    label.config(
                        text=(
                            f"○ {display_name}  "
                            f"({value:.3f} < "
                            f"{threshold:.3f})"
                        )
                    )

        # -------------------------------------------------
        # Feature information
        # -------------------------------------------------

        for key, value in features.items():

            if key not in self.feature_labels:
                continue

            if isinstance(value, float):

                display_value = (
                    f"{value:.4f}"
                )

            else:

                display_value = str(
                    value
                )

            self.feature_labels[
                key
            ].config(
                text=display_value
            )

from tkinter import ttk


class WindowPanel:
    """
    Displays DNS aggregation windows and
    their rule-based detection results.
    """

    def __init__(
        self,
        parent,
        selection_callback
    ):

        self.selection_callback = (
            selection_callback
        )

        self.frame = ttk.LabelFrame(
            parent,
            text="DNS Windows",
            padding=8
        )

        window_columns = (
            "source",
            "domain",
            "start",
            "events",
            "rules",
            "score"
        )

        self.table = ttk.Treeview(
            self.frame,
            columns=window_columns,
            show="headings",
            selectmode="browse"
        )

        window_headings = {
            "source": "Source",
            "domain": "Domain",
            "start": "Start",
            "events": "Events",
            "rules": "Rules",
            "score": "Score"
        }

        window_widths = {
            "source": 115,
            "domain": 180,
            "start": 75,
            "events": 60,
            "rules": 60,
            "score": 65
        }

        for column in window_columns:

            self.table.heading(
                column,
                text=window_headings[column]
            )

            self.table.column(
                column,
                width=window_widths[column],
                anchor="center"
            )

        scrollbar = ttk.Scrollbar(
            self.frame,
            orient="vertical",
            command=self.table.yview
        )

        self.table.configure(
            yscrollcommand=scrollbar.set
        )

        self.table.pack(
            side="left",
            fill="both",
            expand=True
        )

        scrollbar.pack(
            side="right",
            fill="y"
        )

        self.table.bind(
            "<<TreeviewSelect>>",
            self._on_selected
        )

        # Maps unique window key -> Treeview item ID
        self.items = {}

        self,
        key,
        window,
        detection

        """

        if detection is None:

            score = 0.0

        else:

            triggered_count = len(
                detection["triggered_rules"]
            )

            score = detection["score"]

        self.table.item(
            item_id,
            values=(
                window.src_ip,
                window.domain,
                window.start_time.strftime(
                    "%H:%M:%S"
                ),
                window.event_count,
                f"{triggered_count}/5",
                f"{score:.1f}"
            )
        )

    def select_window(self, key):
        """
        Select a window programmatically.
        """

        if key not in self.items:
            return

        item_id = self.items[key]

        self.table.selection_set(
            item_id
        )

        self.table.focus(
            item_id

        if not selection:

import tkinter as tk

class KaliTestPanel:

        self.start_callback = start_callback

        # -------------------------------------------------
        # Interface
        # -------------------------------------------------

        tk.Label(
            self.frame,
            text="TShark Interface:"
        ).pack(side="left", padx=(0, 5))

        self.interface_var = tk.StringVar(
            value="6"
        )

        self.interface_entry = ttk.Entry(
            self.frame,
            textvariable=self.interface_var,
            width=8
        )

        self.interface_entry.pack(
            side="left",
            padx=5
        )

        # -------------------------------------------------
        # Start button
        # -------------------------------------------------

        self.start_button = ttk.Button(
            self.frame,
            text="Start Kali Test",
            command=self.start_test
        )

        self.start_button.pack(
            side="left",
            padx=5
        )

        # -------------------------------------------------
        # Stop button
        # -------------------------------------------------

        self.stop_button = ttk.Button(
            self.frame,
            text="Stop Kali Test",
            command=self.stop_test,
            state="disabled"
        )

        self.stop_button.pack(
            side="left",
            padx=5
        )

    def get_interface(self):
        value = self.interface_var.get().strip()

        if not value:
            raise ValueError(
                "Kali test interface cannot be empty."
            )

        try:
            interface = int(value)
        except ValueError:
            raise ValueError(
                "Kali test interface must be a number."
            )

        if interface <= 0:
            raise ValueError(
                "Kali test interface must be positive."
            )

        return interface

    def start_test(self):
        self.start_callback(
            self.get_interface()
        )

    def stop_test(self):
        self.stop_callback()

    def set_running(self):
        self.start_button.config(
            state="disabled"
        )

        self.stop_button.config(
            state="normal"
        )

        self.interface_entry.config(
            state="disabled"
        )

    def set_stopped(self):
        self.start_button.config(

        )

            state="disabled"
        )
